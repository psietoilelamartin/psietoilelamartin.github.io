\begin{enumerate}
\item
Soit $P$ une probabilité sur $(\mathbb{N}, \mathscr{P}(\mathbb{N}))$.
\begin{enumerate}
\item Montrer que $P(\{n\}) \xrightarrow[n \rightarrow+\infty]{ } 0$.
\item Que peut-on dire de $\dlim _{n \rightarrow+\infty} P(\ens{k\in\N,\ k \geqslant n})$ ?
\end{enumerate}
\item Soit $\left(a_n\right)_{n \in \mathbb{N}}$ une suite strictement décroissante de réels positifs de limite nulle.
Déterminer $\lambda \in \mathbb{R}$ tel qu'il existe une probabilité $P$ sur $(\mathbb{N}, \mathscr{P}(\mathbb{N}))$ vérifiant
$\forall\, n\in\N$, $P(\ens{k\in\N,\ k \geqslant n})=\lambda a_n$.
\end{enumerate}
